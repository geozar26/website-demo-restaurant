    <script>
        // Elements
        const fab = document.getElementById('fab');
        const tooltipCard = document.getElementById('tooltipCard');
        const overlay = document.getElementById('overlay');
        const decreaseBtn = document.getElementById('decreaseBtn');
        const increaseBtn = document.getElementById('increaseBtn');
        const quantityValue = document.getElementById('quantityValue');
        const orderBtn = document.getElementById('orderBtn');

        let quantity = 1;
        let isOpen = false;

        // Toggle tooltip
        function toggleTooltip() {
            isOpen = !isOpen;
            
            if (isOpen) {
                tooltipCard.classList.add('active');
                fab.classList.add('active');
                overlay.classList.add('active');
            } else {
                tooltipCard.classList.remove('active');
                fab.classList.remove('active');
                overlay.classList.remove('active');
            }
        }

        // Close tooltip
        function closeTooltip() {
            isOpen = false;
            tooltipCard.classList.remove('active');
            fab.classList.remove('active');
            overlay.classList.remove('active');
        }

        // Event Listeners
        fab.addEventListener('click', toggleTooltip);
        overlay.addEventListener('click', closeTooltip);

        // Quantity controls
        decreaseBtn.addEventListener('click', () => {
            if (quantity > 1) {
                quantity--;
                quantityValue.textContent = quantity;
                animateQuantityChange();
            }
        });

        increaseBtn.addEventListener('click', () => {
            if (quantity < 99) {
                quantity++;
                quantityValue.textContent = quantity;
                animateQuantityChange();
            }
        });

        function animateQuantityChange() {
            quantityValue.style.transform = 'scale(1.3)';
            setTimeout(() => {
                quantityValue.style.transform = 'scale(1)';
            }, 200);
        }

        // Order button
        orderBtn.addEventListener('click', () => {
            orderBtn.classList.add('success');
            orderBtn.textContent = '✓ Επιτυχία!';
            
            setTimeout(() => {
                closeTooltip();
                setTimeout(() => {
                    orderBtn.classList.remove('success');
                    orderBtn.textContent = 'Παραγγελία';
                    quantity = 1;
                    quantityValue.textContent = quantity;
                }, 300);
            }, 1500);
        });

        // Smooth transitions for quantity
        quantityValue.style.transition = 'transform 0.2s cubic-bezier(0.34, 1.56, 0.64, 1)';

        // Prevent card click from closing
        tooltipCard.addEventListener('click', (e) => {
            e.stopPropagation();
        });

        // Keyboard support
        document.addEventListener('keydown', (e) => {
            if (e.key === 'Escape' && isOpen) {
                closeTooltip();
            }
        });
    </script>
