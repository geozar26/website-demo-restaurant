document.addEventListener('DOMContentLoaded', () => {
    // Επιλέγουμε όλα τα εικονίδια
    const icons = document.querySelectorAll('.small-icon');

    icons.forEach(icon => {
        icon.addEventListener('click', (e) => {
            e.stopPropagation();
            
            // Βρίσκουμε το tooltip που ανήκει στην ΙΔΙΑ κάρτα με το εικονίδιο
            const parent = icon.parentElement;
            const tooltip = parent.querySelector('.carousel-tooltip');

            if (!tooltip.classList.contains('active')) {
                // ΑΝΟΙΓΜΑ
                tooltip.classList.remove('closing');
                tooltip.classList.add('active');
            } else {
                // ΚΛΕΙΣΙΜΟ (με το animation organic-shrink)
                tooltip.classList.add('closing');
                
                // Περιμένουμε να τελειώσει το animation (0.4s) πριν βγάλουμε την κλάση active
                setTimeout(() => {
                    tooltip.classList.remove('active');
                    tooltip.classList.remove('closing');
                }, 400); 
            }
        });
    });

    // Κλείσιμο αν πατήσεις οπουδήποτε αλλού στη σελίδα
    document.addEventListener('click', () => {
        const activeTooltips = document.querySelectorAll('.carousel-tooltip.active');
        activeTooltips.forEach(tooltip => {
            tooltip.classList.add('closing');
            setTimeout(() => {
                tooltip.classList.remove('active');
                tooltip.classList.remove('closing');
            }, 400);
        });
    });
});
