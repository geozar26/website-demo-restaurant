2. Υλοποίηση του initGSAP() (JavaScript)
Μέσα στην κλάση σου RestaurantApp, αντικατέστησε το κενό (ή το υπάρχον) initGSAP() με το παρακάτω. Έχω σχεδιάσει διαφορετικό animation για κάθε section:

JavaScript
initGSAP() {
    // Καταχωρούμε το ScrollTrigger plugin
    gsap.registerPlugin(ScrollTrigger);

    // --- SECTION: ΠΡΟΤΑΣΕΙΣ ΓΕΥΣΕΩΝ (.items) ---
    // Animation: Slide up & Fade in με "stagger"
    gsap.from(".items .item", {
        scrollTrigger: {
            trigger: ".items",
            start: "top 80%", // Ξεκινάει όταν το section φανεί στο 80% της οθόνης
        },
        y: 100,          // Ξεκινάει από 100px πιο κάτω
        opacity: 0,
        duration: 1,
        ease: "power2.out",
        stagger: 0.15    // Κάθε πιάτο εμφανίζεται 0.15 δευτερόλεπτα μετά το προηγούμενο
    });

    // --- SECTION: TESTIMONIALS (.testimonials-section) ---
    // Animation: Scale up & Rotate ελαφρά (Pop effect)
    gsap.from(".testimonial-card", {
        scrollTrigger: {
            trigger: ".testimonials-section",
            start: "top 75%",
        },
        scale: 0.8,
        opacity: 0,
        rotation: 2,
        duration: 1.2,
        ease: "back.out(1.7)", // Δίνει μια ελαφριά αναπήδηση στο τέλος
        stagger: 0.3
    });

    // --- SECTION: GALLERY (Αν υπάρχει κλάση .gallery-section) ---
    // Animation: Reveal από τα πλάγια (αριστερά/δεξιά εναλλάξ)
    gsap.from(".gallery-section .carousel-card", {
        scrollTrigger: {
            trigger: ".gallery-section",
            start: "top 80%",
        },
        x: (index) => index % 2 === 0 ? -100 : 100, // Τα ζυγά από αριστερά, τα μονά από δεξιά
        opacity: 0,
        duration: 1,
        ease: "power3.out",
        stagger: 0.2
    });

    // --- SECTION: FOOTER ---
    // Animation: Τα στοιχεία του footer "ξεδιπλώνονται" από κάτω
    gsap.from(".nav-footer > *", {
        scrollTrigger: {
            trigger: ".nav-footer",
            start: "top 95%",
        },
        y: 30,
        opacity: 0,
        duration: 0.8,
        ease: "power1.inOut",
        stagger: 0.1
    });
}
